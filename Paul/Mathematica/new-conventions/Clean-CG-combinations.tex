\section{Setup and the question}

\paragraph{Conventions.} These are the new conventions with Hermitian generators $EH$:
\begin{itemize}
\item Lax operator $\mathcal L^{(s)}_{ij}(u)=(u-\theta_s)\delta_{ij}+hE^{(s)}_{ij}$, and monodromy $T(u)=\mathcal L^{(1)}(u)\mathcal L^{(2)}(u)\,\mathrm{diag}(z)$.
\item $t_a$ is the sum of the principal quantum minors.
\item CBR fusion: $T_\lambda(u)=\det_{i,j}t_{\lambda'_j+i-j}(u+(i-1)h)$, with $t_0=1$.
\item $\nu_n(x)=\prod_s(x-\theta_s+h\Lambda_n)$.
\item Direct and dual Q-functions: $Q_i=z_i^{-u/h}q_i$ with $\deg q_i=2\Lambda_1-\mu_i$, and $Q^j=z_j^{u/h}q^j$ with $\deg q^j=\mu_j$ (here $\Lambda_3=0$).
\end{itemize}

\paragraph{The bra.} The bra is $\bra{\Om}=\bra{\Om_{zw}}\otimes\bra{\Om_{zw}}$, the vacuum of the companion $w$-twist written in the frame where $z$ is diagonal. Explicitly $\bra{\Om_{zw}}=e_1^TR$, with
\[
R=e^{r_1E_{12}}e^{r_2E_{13}}e^{r_3E_{23}},\qquad
r_1=\frac{w_1-z_1}{w_1-z_2},\quad
r_2=\frac{(w_1-z_1)(z_1-w_2)z_{23}}{z_{12}(w_1-z_3)(w_2-z_3)},\quad
r_3=\frac{(w_1-z_2)(z_2-w_2)z_{13}}{z_{12}(w_1-z_3)(z_3-w_2)}.
\]
The $r_i$ do not depend on the representation, and $w_3$ drops out of them.

\paragraph{Normalization and twists.} States are normalized by $\langle x_0|\Psi\rangle=1$, where $x_0=(e_1^TP_{\rm phys}^{-1})^{\otimes2}$. The twists are small: $z=e^{\eps\varphi}$, $w=e^{\eps\omega}$. HW-limit states have $\Psi_B\to\eps^{-k}h_\mu$, with $h_\mu$ the HW CG vector.

\paragraph{Q-side data.} For any insertion $X$ built from $t$'s,
\[
\bra{\Om}X\ket{\Psi_B}=X^B\langle\Om|\Psi_B\rangle .
\]
Here $\langle\Om|\Psi_B\rangle$ is the $6\times6$ determinant formula of \texttt{L2-diagonal-frame-overlap.md}. That formula is validated for the adjoint only.

\paragraph{The question.} Can a closed Q-expression give the CG data directly, without reconstructing the whole eigenvector? Earlier work used the localized site-1 insertions $\bra\Om T_\lambda(\theta_1-\Lambda_1h)=f_\lambda\,\bra{v_\lambda}\otimes\bra{\Om_{zw}}$.

\section{The contamination}

\subsection{Mechanism (adjoint; \texttt{Minimal-ingredients-EH.wb}, Sections 5--6)}
Undoing $R$ gives $\bra\Om T_\lambda(\theta_1-2h)=f_\lambda\,c_\lambda(R\otimes R)$, with $c_\lambda=\bra{v_\lambda}R^{-1}\otimes e_1^T$. Since raising operators kill HW vectors, $(R\otimes R)P_\HW=P_\HW$ exactly. This gives an exact split at any $\eps$:
\[
\frac{T^B_\lambda\langle\Om|\Psi_B\rangle}{f_\lambda}
=\underbrace{c_\lambda\cdot P_\HW\Psi}_{\text{signal }S}
+\underbrace{c_\lambda\cdot(R\otimes R)(1-P_\HW)\Psi}_{\text{contamination }K},
\qquad \kappa=\frac{S+K}{S}.
\]

\paragraph{Why it is first-order perturbation theory.} At $z=1$ the twist is trivial, so $c_\lambda\propto e_1\otimes e_1$, which is pure top weight. Its components on lower weights are $O(\eps)$ (factors $z_{ij}$). Hence:
\begin{itemize}
\item the signal $c\cdot h_\mu$ is small;
\item the $O(1)$ top-weight part of $c$ meets the $O(\eps)$ correction $\psi_1$ of the eigenvector, which $R$ raises to the top weight;
\item writing $c=a_0+\eps a_1$ and $\Psi=\psi_0+\eps\psi_1$ with $a_0\cdot\psi_0=0$, one gets $c\cdot\Psi=\eps(a_1\psi_0+a_0\psi_1)$.
\end{itemize}

\paragraph{Measured values.}
\begin{itemize}
\item $\kappa=\tfrac{t+1}{2}=\tfrac{97}{70}$ for $[3,3,0]$ and for $[4,1,1]$ with $(2,1)$.
\item Octets: $(1.530,\,1.072)$ and $(0.642,\,0.772)$.
\item Singlet with $(2,1)$: $\kappa\approx1.102=\tfrac{(t+1)(t+3)}{12}$.
\item For $[3,3,0]$ the whole contamination is the $\nu=[4,2,0]$ term, with $\|c^{(420)}\|\sim\eps^0$ and $\|[(R\otimes R)\Psi_\perp]_{420}\|\sim\eps^{-2}$, against a signal $\eps^1\cdot\eps^{-3}$. The term divided by the signal is $\tfrac{27}{70}=\kappa-1$.
\end{itemize}

\subsection{Why projectors and $w$-expansions cannot remove it}
These are arguments; the \texttt{[3,3,0]} counterexample is numerical.
\begin{itemize}
\item \textbf{Weight projectors.} $T_\lambda$ commutes with the Cartan and $\Psi$ has a fixed weight, so a weight projector on the left-hand side is trivial. It is nontrivial only in the frame with $R$ undone, where it would require $(R\otimes R)^{-1}$ to act on $\Psi$, and that is not a Q-quantity.
\item \textbf{Torus degree.} Under the torus acting on $(r_1,r_2,r_3)$ with weights $(\alpha_1,\alpha_1+\alpha_2,\alpha_2)$, the component $c^{(\nu)}$ has degree $\mathrm{top}-\nu$ and $[(R\otimes R)\Psi]_\nu$ has degree $\nu-\mu$. Every term in $\sum_\nu c^{(\nu)}\cdot[(R\otimes R)\Psi]_\nu$ therefore has the same degree $\mathrm{top}-\mu$, and no expansion in $w$ separates the CG term from the $[4,2,0]$ term.
\item \textbf{Counterexample.} For $[3,3,0]$ the left-hand side is exactly $\propto r_1$, a single monomial, and its coefficient is $\kappa\times$CG.
\item \textbf{A third independent $r$ adds nothing.} Modulo the torus there is a single invariant, $\rho=r_2/(r_1r_3)=-\frac{(z_1-w_2)z_{23}}{(z_2-w_2)z_{13}}$. It already varies with $w_2$, so the missing direction is a torus direction, which acts by a known scale.
\item \textbf{Coherent states see only the top irrep.} Any coherent bra $(e_1^Tg)^{\otimes2}$ lies in the dual of the top irrep. All access to other irreps comes from the insertions.
\end{itemize}

\section{Clean combinations}

\subsection{Top-killing is exact, and is an eigenvalue difference}
For $\mu\neq\mathrm{top}$ we have $e_1e_1\cdot P_\HW\Psi=0$, so
\[
\langle\Om|\Psi\rangle=e_1e_1\cdot(R\otimes R)\Psi_\perp .
\]
The top-weight contamination of $c_\lambda$ is therefore exactly $(c_\lambda\cdot e_1e_1)\langle\Om|\Psi\rangle$. In the bare frame $c_\lambda\cdot e_1e_1=\bra\Om X\ket{e_1e_1}=X^{\rm top}$, the eigenvalue on the top state. So the top-killed Q-number of $X$ is
\[
\boxed{(X^B-X^{\rm top})\,\langle\Om|\Psi_B\rangle .}
\]
In the bare frame, $S_X=W_X\cdot P_\HW\Psi$ and $K_X=W_X\cdot\Psi_\perp-X^{\rm top}\langle\Om|\Psi\rangle$ with $W_X=\bra\Om X$, so no $R\otimes R$ is needed. This holds for any representation.

\paragraph{Clean combinations.} For each weight sector, collect the leading-order ratios $k_B=K/S_{\rm ref}$ of every HW-limit copy $B$. The clean combinations are the null space of these. For all three representations below the $k_B$ were found to be \emph{structural}: independent of $\varphi$ and $\omega$ up to $O(\eps)$. The HW part of $\Psi$, scale included, then follows by least squares over several $w$.

\subsection{$[4,2,0]^2$: the blind spot was wrong (\texttt{Blindspot-420-check-EH.wb})}
\texttt{L2-420-localization.md} claimed that $\HW_{[5,4,3]}$ and $\HW_{[4,4,4]}$ lie exactly in the null space of the localized data. That claim is wrong.
\begin{itemize}
\item A site-1-local covector is $u\otimes e_1^T$, and $c\cdot h=u\cdot(1\otimes e_1^T)h$. The slice $(1\otimes e_1^T)h$ is never zero, because the largest site-2 weight in $h$ must be the site-2 highest weight.
\item $T_{(3,2)}$ and $T_{(4,2)}$ have components at site-1 weight $(1,2,3)$, and $T_{(4,2)}$ also at $(0,2,4)$.
\item The overlaps are $8\times10^{-3}$ on $[5,4,3]$ and $8\times10^{-4}$ on $[4,4,4]$ at the finite twist. At small twist they scale exactly as $\eps^3$ and $\eps^4$, while $\|u\|\approx2.5$.
\item The old ``overlap singular value $=1.000$'' was printed at 3 digits and only shows near-containment in a numerical null space.
\end{itemize}

\section{The site-1 eigenvalues as single Q-functions}

\subsection{Reduction to $2\times2$ determinants}
Let $v(u)=(Q^1,Q^2,Q^3)(u)$. The quantum Weyl formula reads
\[
T_\lambda(x)=P_\lambda(x)\,\frac{\det[v(x+\lambda_1h),v(x+(\lambda_2-1)h),v(x-2h)]}{\det[v(x),v(x-h),v(x-2h)]},
\qquad P_\lambda(x)=\prod_{(r,c)\in\lambda}\nu_3(x+(c-r)h).
\]
Two degeneracies hold near $\theta_1$:
\begin{itemize}
\item \textbf{3$\times$3 Casoratian.} The dual Baxter equation fixes $W(x)/W(x-h)=\chi_3\nu_1(x)\nu_2(x-h)/(\nu_3(x)\nu_3(x-h))$. Hence $\det[v(x),v(x-h),v(x-2h)]$ vanishes for $x=\theta_s+h,\dots,\theta_s-(\Lambda_1-1)h$.
\item \textbf{Dual parallelism} $v(\theta_1-\Lambda_2h)\parallel\dots\parallel v(\theta_1)$, from the dual QQ relation below.
\end{itemize}
Together these put $v(\theta_1-(\Lambda_1+1)h),\dots,v(\theta_1)$ in one 2-plane, and
\[
\boxed{T_\lambda(\theta_1-\Lambda_1h)=P_\lambda\,\frac{\det_2[v(\theta_1+(\lambda_1-\Lambda_1)h),\,v(\theta_1+(\lambda_2-\Lambda_1-1)h)]}{\det_2[v(\theta_1-\Lambda_1h),\,v(\theta_1-(\Lambda_1+1)h)]}}\qquad(\lambda_1\le\Lambda_1,\ \lambda_3=0),
\]
with any fixed pair of rows. This is the $L=2$ version of the $L=1$ ``reduction at $\theta-2h$''. It was checked exactly on 34 states of $[3,1,0]^2$ (to $10^{-70}$) and on 91 states of $[4,2,0]^2$ (to $10^{-194}$), for all shapes and every row pair.

\subsection{The dual QQ relation and single Q's}
For $[4,2,0]$, all 91 states satisfy, to at least 226 digits (the shifts $0$ and $-1$ fail):
\[
\boxed{z_b\,q^a(u)q^b(u+h)-z_a\,q^b(u)q^a(u+h)=(z_b-z_a)\,G(u)\,q_c(u+h),\qquad G(u)=\prod_s\prod_{n=1}^{\Lambda_2}(u-\theta_s+nh).}
\]
As a consequence, the direct vectors $(Q_1,Q_2,Q_3)$ are parallel at $\theta_1-4h$, $\theta_1-3h$ and $\theta_1-2h$, and each consecutive $\det_2$ becomes a single direct $Q$. The localization factors $f_\lambda$ cancel all $\tw$-dependence.

\paragraph{$[4,2,0]$ at $\theta_1-4h$} (exact to $10^{-225}$, all $c$ and $j$):

\frac{T_{(2,2)}}{f_{(2,2)}}=40h^4\chi_3^2\,r_1r_2,\qquad

with $r_1=\frac{Q_c(\theta_1-3h)}{Q_c(\theta_1-4h)}$, $r_2=\frac{Q_c(\theta_1-2h)}{Q_c(\theta_1-3h)}$, $\rho_k=\frac{Q^j(\theta_1-kh)}{Q^j(\theta_1-2h)}$.

\section{The adjoint clean combination, derived analytically}\label{sec:adjoint}
\texttt{Adjoint-clean-analytic.wb} works exactly in the $E$ basis with $t$ symbolic and $h=1$; $h$ is restored by homogeneity.

\paragraph{Mechanism.} $\bra{\Om(\eps)}$ lies in the 27, on which the untwisted $T_\lambda$ acts by $T^{\rm top}_\lambda$. Therefore $(W_\lambda-T^{\rm top}_\lambda\Om)$ vanishes at $\eps^0$ on the weight sector; this is checked. With $\Psi=h+\eps\psi_1+\dots$,
\[
S_\lambda=\Om(\eps)\cdot T_\lambda(\eps)\,h,\qquad K_\lambda=\eps^2\,(W_\lambda-T^{\rm top}_\lambda\Om)_1\cdot\psi_1+O(\eps^3),
\]
and only first-order perturbation theory is needed.
\begin{itemize}
\item $[4,1,1]$, $[3,3,0]$ and the octets have signal $O(\eps)$, so they are clean automatically.
\item The singlet has $S_{(1,1)}\equiv0$ and $S_{(2,1)}=O(\eps^2)$. In raw eigenvalues, exactly and for both direction pairs tested,
\[
K_{(1,1)}/S_{(2,1)}=-\tfrac13,\qquad K_{(2,1)}/S_{(2,1)}=\tfrac{2t-3}{3}.
\]
\end{itemize}

\paragraph{Result.} A single combination is clean for all six HW states:
\[
\boxed{\mathcal C=T_{(2,1)}(\theta_1-2h)+h(2\tw-3h)\,T_{(1,1)}(\theta_1-2h),\qquad\text{Q-number}=(\mathcal C^B-\mathcal C^{\rm top})\langle\Om|\Psi_B\rangle .}
\]
This is $\beta=h(2t-3)/(t-1)$ in the $f$-normalized basis, matching the fit. In Q's, dropping the state-independent factor $3h^3\chi_3(\tw-h)(\tw-3h)$:
\[
\Big\{r\big[(2\tw-3h)-(\tw-h)\rho\big]-z_1^{-1}\big[(2\tw-3h)-(\tw-h)z_3\big]\Big\}\,\langle\Om|\Psi_B\rangle .
\]

\paragraph{Rescaling the monodromy.} Under $T\to T/g$, the coefficient changes only through $g(\theta_1-h)$. The resulting clean coefficient $c$ in $\tilde T_{(2,1)}+c\,\tilde T_{(1,1)}$ is:
\begin{center}
\begin{tabular}{lcccccc}
\toprule
$g$ & none & $\nu_1$ & $\nu_2$ & $\nu_3$ & $f$-norm & $\prod_s(u-\theta_s-h/2)$\\
$c$ & $2t-3$ & $\frac{2t-3}{t+1}$ & singular & $-\frac{2t-3}{t-1}$ & $\frac{2t-3}{t-1}$ & $-\tfrac43$\\
\bottomrule
\end{tabular}

The half-shifted rescaling, i.e.\ each Lax divided by $(u-\theta_s-h/2)$, gives the $t$-free combination $\tilde T_{(2,1)}-\frac43\tilde T_{(1,1)}$. Among products of site factors, the site-2 half shift is forced.

\section{Numerical traps found today}
\begin{itemize}
\item \textbf{Fake zeros.} A zero with low accuracy prints as an ``exact'' 0. At 120 digits the $[4,4,4]$ errors at $\eps\le10^{-5}$ had no correct digits; always report \texttt{Precision}. Taking \texttt{Log10} of such a zero gives \texttt{Indeterminate}.
\item \textbf{Already-clean insertions.} Such an insertion has $k=O(\eps)$. A relative null-space cut treats that noise as a constraint and discards the useful insertion; use an absolute cut (\texttt{kTol}).
\item \textbf{List-minus-vector threading.} Subtracting a vector from a list of vectors whose length happens to match threads element by element and gives a fake ``spread''.
\item \textbf{Ill-conditioned $x_0$.} \texttt{Inverse}$[P_{\rm phys}]$ is badly conditioned at small $\eps$. Use the factorized inverse.
\item \textbf{Nonzero-entry lookup.} \texttt{Position} with a pattern test returned position 0 (the head), which silently set $m_0=1$. Use \texttt{Ordering[Abs[h],-1]}.
\item \textbf{Slow symbolic substitution.} Substituting symbolic $64\times64$ operators per state is far too slow. Substitute once and cache.
\item \textbf{Shared kernel.} All new-conventions notebooks share one kernel and clear it on their first cell, so rerun a notebook before resuming work in it.
\end{itemize}
